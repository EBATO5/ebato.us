  function advanceCoinWithLinearFriction(c,dt){
    const speed=Math.hypot(c.vx,c.vy);
    if(speed<=0){c.vx=0;c.vy=0;return}

    const nextSpeed=Math.max(0,speed-COIN_FRICTION_DECEL*dt);
    // Use average speed over the substep so distance stays accurate as the coin slows.
    const travelSpeed=(speed+nextSpeed)*.5;
    const nx=c.vx/speed,ny=c.vy/speed;
    c.x+=nx*travelSpeed*dt;
    c.y+=ny*travelSpeed*dt;
    c.vx=nx*nextSpeed;
    c.vy=ny*nextSpeed;
  }
