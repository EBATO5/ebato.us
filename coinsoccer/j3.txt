mewhat more likely, but the AI intentionally
    // samples among near-equal plans instead of deterministically taking #1.
    const weights=pool.map(c=>Math.exp((c.value-best)/10)+.18);
    let roll=Math.random()*weights.reduce((a,b)=>a+b,0);
    for(let i=0;i<pool.length;i++){
      roll-=weights[i];
      if(roll<=0)return pool[i];
    }

  function aiKickoffShot(){
    // One recognizable break: drive the rear/center coin into the seam so it
    // contacts both front coins in the same physics step. Variation comes from
    // strength plus only a hair of left/right angle, not from changing the play.
    const index=1;
    const c=coins[index];
    const angle=(Math.random()*2-1)*.0045;
    const power=.30+Math.random()*.18;
    const speed=shotSpeedForPower(power);
    const ux=Math.sin(angle);
    const uy=Math.cos(angle); // AI/P2 attacks downward in canonical coordinates.

    aiLastKickoffFamily='center-break';
    aiLastKickoffPattern=`center:${Math.round(power*100)}:${angle>=0?'R':'L'}`;
    aiLastOpeningPattern='';
    aiLastCoin=index;
    aiShotCount++;
    aiPossessionShots++;
    beginShot(index,ux*speed,uy*speed);
  }

  function aiOpeningContinuationShot(){
    // The kickoff can leave two coins touching. Keep the anti-stuck formation
    // check, but choose among COINS first so one coin cannot flood the candidate
    // pool with many tiny variations of the same move.
    const choices=[];

    for(let i=0;i<coins.length;i++){
      const c=coins[i];

      for(const cand of aiCandidateTargets(i)){
        const predicted=coins.map(o=>({...o,vx:0,vy:0}));
        predicted[i]={...predicted[i],x:cand.target.x,y:cand.target.y};
        const qualities=predicted.map((_,j)=>aiNextGateQuality(predicted,j));
        const minQuality=Math.min(...qualities);
        const avgQuality=qualities.reduce((a,b)=>a+b,0)/qualities.length;
        const progress=(cand.target.y-c.y)/field.h;

        let value=minQuality*3.2 + avgQuality*.8 + cand.value*.18 + progress*14;
        if(minQuality<20)value-=220;
        choices.push({...cand,value,minQuality,avgQuality});
      }
    }

    if(!choices.length)return false;

    choices.sort((a,b)=>b.value-a.value);
    const bestMin=Math.max(...choices.map(c=>c.minQuality??-80));
    let safeChoices=choices.filter(c=>(c.minQuality??-80)>=Math.max(30,bestMin-12));
    if(!safeChoices.length)safeChoices=choices;
    safeChoices.forEach(c=>{
      const tx=(c.target.x-field.x)/Math.max(1,field.w);
      const ty=(c.target.y-field.y)/Math.max(1,field.h);
      c.patternKey=`${aiPossessionShots}:${c.index}:${tx.toFixed(2)}:${ty.toFixed(2)}`;
    });

    const byCoin=new Map();
    for(const cand of safeChoices){
      const bucket=byCoin.get(cand.index)||[];
      bucket.push(cand);
      byCoin.set(cand.index,bucket);
    }

    const coinPlans=[];
    for(const bucket of byCoin.values()){
      bucket.sort((a,b)=>b.value-a.value);
      const bestPlan=bucket[0];
      const targetPlan=aiVariedChoice(bucket,18,6,aiLastOpeningPattern)||bestPlan;
      coinPlans.push({...targetPlan,coinValue:bestPlan.value});
    }

    coinPlans.sort((a,b)=>b.coinValue-a.coinValue);
    const bestCoinValue=coinPlans[0].coinValue;
    const dimePlan=coinPlans.find(c=>c.c.type==='dime');
    let choice=null;

    // After the kickoff, a near-tie should favor advancing the 2-point coin.
    // Only send the untouched penny next when it offers a materially better plan.
    if(aiPossessionShots===1 && dimePlan && dimePlan.coinValue>=bestCoinValue-10){
      choice=dimePlan;
    }else{
      const coinPool=coinPlans.filter(c=>c.coinValue>=bestCoinValue-45);
      const coinWeights=coinPool.map(c=>
        Math.exp((c.coinValue-bestCoinValue)/22)+.55+(c.c.type==='dime'?.08:0)
      );
      let coinRoll=Math.random()*coinWeights.reduce((a,b)=>a+b,0);
      choice=coinPool[coinPool.length-1];
      for(let i=0;i<coinPool.length;i++){
        coinRoll-=coinWeights[i];
        if(coinRoll<=0){choice=coinPool[i];break;}
      }
    }

    aiLastOpeningPattern=choice.patternKey||'';
    const dist=Math.hypot(choice.target.x-choice.c.x,choice.target.y-choice.c.y);
    const power=aiPowerForDistance(dist,.16,.66);
    aiAimAndShoot(choice.index,choice.target,choice.gate,power);
    return true;
  }

  function aiChooseShot(){
    if(kickoffShot){
      aiKickoffShot();
      return;
    }

    if(aiPossessionShots===1 || aiPossessionShots===2){
      if(aiOpeningContinuationShot())return;
    }

    aiLastProgress=aiTeamProgress(coins);

    const scoringShots=[];
    for(let i=0;i<coins.length;i++){
      const c=coins[i];
      const gate=aiGateFor(i);
      const range=aiSafeCrossRange(c,gate);
      if(!range)continue;

      for(const target of aiGoalTargets(c)){
        if(!aiRayFitsGate(c,target,gate))continue;

        const dx=target.x-c.x,dy=target.y-c.y;
        const dist=Math.hypot(dx,dy);
        const points=coinPointValue(c);
        const widthComfort=aiClamp(gate.opening/(c.r*3),0,1);
        const keeperComfort=aiClamp((target.keeperClearance||0)/Math.max(1,field.goalWidth*.45),0,1);
        const value=points*115 + widthComfort*28 + keeperComfort*36 - dist/field.h*10;

        scoringShots.push({
          index:i,c,gate,target,
          power:aiPowerForDistance(dist,.30,.80),
          value
        });
        break;
      }
    }

    if(scoringShots.length){
      scoringShots.sort((a,b)=>b.value-a.value);
      const s=scoringShots[0];
      aiAimAndShoot(s.index,s.target,s.gate,s.power);
      return;
    }

    const candidates=[];
    for(let i=0;i<coins.length;i++){
      candidates.push(...aiCandidateTargets(i));
    }

    if(candidates.length){
      candidates.sort((a,b)=>b.value-a.value);
      const choice=candidates[0];
      const dist=Math.hypot(choice.target.x-choice.c.x,choice.target.y-choice.c.y);
      const power=aiPowerForDistance(dist,.18,.70);
      aiAimAndShoot(choice.index,choice.target,choice.gate,power);
      return;
    }

    let recovery=null;
    for(let i=0;i<coins.length;i++){
      const c=coins[i];
      const gate=aiGateFor(i);
      const score=gate.opening;
      if(!recovery||score>recovery.score)recovery={i,c,gate,score};
    }

    if(recovery){
      const {i,c,gate}=recovery;
      let dx=gate.mx-c.x,dy=gate.my-c.y;
      const len=Math.hypot(dx,dy)||1;
      dx/=len;dy/=len;

      const target={
        x:aiClamp(gate.mx+dx*field.h*.11,field.x+c.r+2,field.x+field.w-c.r-2),
        y:aiClamp(gate.my+dy*field.h*.11,field.y+c.r+2,field.y+field.h-c.r-2)
      };
      const dist=Math.hypot(target.x-c.x,target.y-c.y);
      aiAimAndShoot(i,target,gate,aiPowerForDistance(dist,.18,.58));
    }
  }
