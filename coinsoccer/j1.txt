attacker===0;
    if(opponentMode==='online')return onlineConnected&&attacker===onlineRole;
    return true;
  }

  function canControlKeeperLocal(){
    if(!goalie)return false;
    if(opponentMode==='online')return onlineConnected&&goalie.side===onlineRole;
    if(opponentMode==='ai')return goalie.side===0;
    return true;
  }

  canvas.addEventListener('pointerdown',e=>{
    canvas.setPointerCapture(e.pointerId);
    const p=posFromEvent(e);

    if(canControlKeeperLocal()&&hitGoalie(p)){
      pointer={id:e.pointerId,mode:'goalie'};
      if(onlineActive()&&onlineRole===1){
        sendBroadcast('keeper_hold',{player:1,held:true});
      }
      return;
    }

    if(!canShootLocal())return;

    const i=hitCoin(p);
    if(i>=0){
      selected=i;
      pointer={id:e.pointerId,mode:'aim'};
      aim=p;
      hintEl.textContent=`${coins[i].type==='dime'?'Dime':'Penny'} selected — pull backward, release to shoot.`;
    }
  });

  canvas.addEventListener('pointermove',e=>{
    if(!pointer||e.pointerId!==pointer.id)return;
    const p=posFromEvent(e);

    if(pointer.mode==='goalie'){
      const left=field.x+(field.w-field.goalWidth)/2+goalie.r;
      const right=field.x+(field.w+field.goalWidth)/2-goalie.r;
      goalie.x=Math.max(left,Math.min(right,p.x));

      if(onlineActive()&&onlineRole===1){
        const now=performance.now();
        if(now-keeperSendAt>28){
          keeperSendAt=now;
          sendBroadcast('keeper_move',{
            player:onlineRole,
            x:(goalie.x-field.x)/field.w
          });
        }
      }
    }else if(pointer.mode==='aim'){
      aim=p;
    }
  });

  function beginShot(index,vx,vy){
    if(running||index<0||index>=coins.length)return;
    const c=coins[index];
    c.vx=vx;c.vy=vy;

    goalieDir=Math.random()<.5?-1:1;

    const others=[0,1,2].filter(i=>i!==index);
    gateSnapshot={
      a:{x:coins[others[0]].x,y:coins[others[0]].y,r:coins[others[0]].r},
      b:{x:coins[others[1]].x,y:coins[others[1]].y,r:coins[others[1]].r}
    };

    shotCoinIndex=index;
    crossedGate=false;
    running=true;
    shotActive=true;
    hintEl.textContent='Shot in motion…';
    sendNetworkState(true);
  }

  function release(e){
    if(!pointer||e.pointerId!==pointer.id)return;

    if(pointer.mode==='goalie'&&onlineActive()&&onlineRole===1){
      sendBroadcast('keeper_hold',{player:1,held:false});
    }

    if(pointer.mode==='aim'&&selected>=0){
      const c=coins[selected];
      const v=viewPoint(c);
      const dx=v.x-aim.x;
      const dy=v.y-aim.y;
      const len=Math.hypot(dx,dy);

      if(len>8){
        const maxPull=maxPullDistance();
        const pull=Math.min(maxPull,len);
        const power=pull/maxPull;
        const ux=dx/len,uy=dy/len;

        if(onlineActive()&&onlineRole===1){
          sendBroadcast('shot',{
            player:1,index:selected,ux,uy,power
          });
          hintEl.textContent='Shot sent…';
        }else{
          const speed=shotSpeedForPower(power);
          const canonicalUy=mirrorView()?-uy:uy;
          beginShot(selected,ux*speed,canonicalUy*speed);
        }
      }
    }

    pointer=null;
    aim=null;
    selected=-1;
  }

  canvas.addEventListener('pointerup',release);
  canvas.addEventListener('pointercancel',release);

  // Desktop: right-click cancels the current pull without taking a shot.
  canvas.addEventListener('contextmenu',e=>{
    e.preventDefault();
    if(selected>=0 || (pointer && pointer.mode==='aim')){
      selected=-1;
      pointer=null;
      aim=null;
      hintEl.textContent='Shot canceled. Choose a coin.';
      toast('Canceled');
    }
  });

  function checkGate(c){
    if(!gateSnapshot||crossedGate)return;
    const a=gateSnapshot.a,b=gateSnapshot.b;
    const gateLen=Math.hypot(b.x-a.x,b.y-a.y);
    const opening=Math.max(4,gateLen-a.r-b.r);
    if(opening<c.r*1.2)return;

    const abx=b.x-a.x,aby=b.y-a.y;
    const t=((c.x-a.x)*abx+(c.y-a.y)*aby)/(gateLen*gateLen||1);
    if(t<=0||t>=1)return;

    const qx=a.x+abx*t,qy=a.y+aby*t;
    const centerDist=Math.hypot(c.x-qx,c.y-qy);
    if(centerDist<=c.r*.55+4){
      const da=Math.hypot(c.x-a.x,c.y-a.y);
      const db=Math.hypot(c.x-b.x,c.y-b.y);
      if(da>c.r+a.r*.78&&db>c.r+b.r*.78){
        crossedGate=true;
        toast('Gate!');
        broadcastGameEvent('Gate!');
      }
    }
  }

  function resolveCoinCollision(a,b){
    let dx=b.x-a.x,dy=b.y-a.y;
    let d=Math.hypot(dx,dy);
    const min=a.r+b.r;
    if(d===0){dx=.01;dy=0;d=.01}
    if(d>=min)return;

    const nx=dx/d,ny=dy/d;
    const overlap=min-d;
    a.x-=nx*overlap/2;a.y-=ny*overlap/2;
    b.x+=nx*overlap/2;b.y+=ny*overlap/2;

    const rvx=b.vx-a.vx,rvy=b.vy-a.vy;
    const rel=rvx*nx+rvy*ny;
    if(rel<0){
      const restitution=.88;
      const j=-(1+restitution)*rel/2;
      a.vx-=j*nx;a.vy-=j*ny;
      b.vx+=j*nx;b.vy+=j*ny;
    }
  }

  function resolveGoalieCollision(c){
    let dx=c.x-goalie.x,dy=c.y-goalie.y;
    let d=Math.hypot(dx,dy);
    const min=c.r+goalie.r;
    if(d===0){dx=.01;dy=0;d=.01}
    if(d>=min)return;

    const nx=dx/d,ny=dy/d;
    c.x=goalie.x+nx*min;
    c.y=goalie.y+ny*min;
    const vn=c.vx*nx+c.vy*ny;
    if(vn<0){
      c.vx-=1.85*vn*nx;
      c.vy-=1.85*vn*ny;
    }
  }

  function isInsideGoalMouth(x,r=0){
    const left=field.x+(field.w-field.goalWidth)/2;
    const right=field.x+(field.w+field.goalWidth)/2;
    const tolerance=Math.max(3,r*.18);
    return x>left-tolerance&&x<right+tolerance;
  }

  function checkGoal(c){
    if(shotCoinIndex<0||coins[shotCoinIndex]!==c)return false;
    if(!crossedGate)return false;
    if(attacker===0&&c.y<field.y&&isInsideGoalMouth(c.x,c.r))return true;
    if(attacker===1&&c.y>field.y+field.h&&isInsideGoalMouth(c.x,c.r))return true;
    return false;
  }

  function boundaries(c){
    const goalLeft=field.x+(field.w-field.goalWidth)/2;
    const goalRight=field.x+(field.w+field.goalWidth)/2;

    if(c.x-c.r<field.x){c.x=field.x+c.r;c.vx=Math.abs(c.vx)*.72}
    if(c.x+c.r>field.x+field.w){c.x=field.x+field.w-c.r;c.vx=-Math.abs(c.vx)*.72}

    const tol=Math.max(3,c.r*.18);
    const inGoal=c.x>goalLeft-tol&&c.x<goalRight+tol;
    if(c.y-c.r<field.y&&!inGoal){c.y=field.y+c.r;c.vy=Math.abs(c.vy)*.72}
    if(c.y+c.r>field.y+field.h&&!inGoal){c.y=field.y+field.h-c.r;c.vy=-Math.abs(c.vy)*.72}

    const topBack=field.y-field.goalDepth;
    const bottomBack=field.y+field.h+field.goalDepth;
    if(c.y-c.r<topBack){c.y=topBack+c.r;c.vy=Math.abs(c.vy)*.55}
    if(c.y+c.r>bottomBack){c.y=bottomBack-c.r;c.vy=-Math.abs(c.vy)*.55}
  }

  function currentGoalieSpeed(){
    // One baseline AI for now. Easier variants can be derived from this later.
    if(opponentMode==='ai'&&goalie&&goalie.side===1)return 78;
    return goalieSpeed;
  }

  function shouldAutoPatrol(){
    if(!goalie)return false;
    if(pointer && pointer.mode==='goalie' && canControlKeeperLocal())return false;
    if(onlineActive() && isHost() && goalie.side===1 && remoteKeeperHeld)return false;
    return true;
  }

  function step(dt){
    if(!running)return;
    const sub=2,sdt=dt/sub;

    for(let k=0;k<sub;k++){
      if(shouldAutoPatrol()&&(!pointer||pointer.mode!=='goalie')){
        const left=field.x+(field.w-field.goalWidth)/2+goalie.r;
        const right=field.x+(field.w+field.goalWidth)/2-goalie.r;
        goalie.x+=goalieDir*currentGoalieSpeed()*sdt;
        if(goalie.x<=left){goalie.x=left;goalieDir=1}
        else if(goalie.x>=right){goalie.x=right;goalieDir=-1}
      }

      for(const c of coins){
        advanceCoinWithLinearFriction(c,sdt);
        boundaries(c);
      }

      if(shotCoinIndex>=0)checkGate(coins[shotCoinIndex]);

      for(let i=0;i<coins.length;i++){
        for(let j=i+1;j<coins.length;j++)resolveCoinCollision(coins[i],coins[j]);
      }

      for(const c of coins)resolveGoalieCollision(c);

      if(shotCoinIndex>=0&&checkGoal(coins[shotCoinIndex])){
        const scorer=attacker;
        const scoringCoin=coins[shotCoinIndex];

        if(opponentMode==='ai'&&attacker===1&&!kickoffShot){
          aiGateAttempts++;
          aiGateMakes++;
        }
        const points=coinPointValue(scoringCoin);
        const coinName=scoringCoin.type==='dime'?'Dime':'Penny';
        score[scorer]+=points;
        attacker=1-scorer;
        resetFormation();
        updateHUD();
        toast(`${coinName} GOAL! +${points}`);
        broadcastGameEvent(`${coinName} GOAL! Player ${scorer+1} +${points}`);
        sendNetworkState(true);
        return;
      }
    }

    const maxV=Math.max(...coins.map(c=>Math.hypot(c.vx,c.vy)));
    if(maxV<9){
      coins.forEach(c=>{c.vx=0;c.vy=0});
      running=false;
      shotActive=false;

      if(opponentMode==='ai'&&attacker===1&&!kickoffShot){
        aiGateAttempts++;
        if(crossedGate)aiGateMakes++;
      }
